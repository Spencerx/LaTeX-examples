\section*{1. Anschaffung von Split-Klimageräten für städtische Gebäude}

\begin{itemize}
    \item \textbf{Finanzielle Auswirkungen:} \qty{80000}{\EUR} (Finanzhaushalt / Investitionen)
    \item \textbf{Sperrvermerk:} Qualifizierter Sperrvermerk – Freigabe der Mittel erfolgt nach Vorlage des Raum- und Umsetzungskonzepts im Bau- und Umweltausschuss.
    \item \textbf{Maßnahme:} Beschaffung und Installation von hocheffizienten Split-Klimageräten für ca. 20 besonders hitzebelastete Räume.
    \item \textbf{Priorisierung:}
    \begin{enumerate}[label=\arabic*.]
        \item Grundschule (Schulstraße) und Mittelschule (Preysingstraße) – Fokus auf Dachgeschosse und Südseiten.
        \item Bürgerspital (bei verbleibenden Restmitteln)
        \item Gebäude der Stadtwerke (bei weiteren Restmitteln).
    \end{enumerate}
    \item \textbf{Arbeitsauftrag an die Verwaltung:} Ermittlung der vorrangigen Schulräume in Absprache mit den Schulleitungen; Prüfung der BAFA-Kälte-Klima-Richtlinie (Förderung für Geräte mit natürlichen Kältemitteln wie R290/Propan).
    \item \textbf{Begründung:} Extreme Sommerhitze beeinträchtigt die
          Konzentrations- und Leistungsfähigkeit von Schüler*innen und
          Lehrkräften erheblich. Die Maßnahme dient dem Gesundheitsschutz und
          geordneten Lernbedingungen. Die Klimaanlagen können außerdem effizient
          zum Heizen genutzt werden.
\end{itemize}

\section*{2. Mobil-flexibles Nebelsprüh-System zur Hitzeminderung}

\begin{itemize}
    \item \textbf{Finanzielle Auswirkungen:} \qty{8000}{\EUR} (Finanzhaushalt; ca. \qty{5000}{\EUR} Anschaffung, \qty{3000}{\EUR} Installation/Anschluss)
    \item \textbf{Maßnahme:} Beschaffung eines Nebelsprüh-Systems (Orientierung an Systemen wie \textit{Water Oasis} / Vorbild Fürth).
    \item \textbf{Standort:} Hauptstandort Spielplatz Preysingstraße / Sterngasse. Durch mobile Bauweise temporäre Verlegung zu Veranstaltungen möglich.
    \item \textbf{Arbeitsauftrag an die Verwaltung:} Prüfung von Fördermitteln über das bayerische Programm \textit{KommKlimaFöR 2026} (Sonderförderung für klimafreundliche Wasseranlagen zur Kühlung auf Spielplätzen/öffentlichen Plätzen mit bis zu 50 \% Förderquote).
    \item \textbf{Begründung \& Zielgruppe:} Verdunstungskühlung senkt punktuell die Temperatur an heißen Tagen bei minimalem Wasserverbrauch. Die Anlage schützt im Bereich des Spielplatzes täglich zahlreiche Kinder, Familien und Senioren im dicht bebauten Stadtgebiet vor extremer Hitzebelastung.
\end{itemize}

\section*{3. Errichtung eines Igel-Lehrpfades an der Mühlbach-Promenade}

\begin{itemize}
    \item \textbf{Finanzielle Auswirkungen:} \qty{1000}{\EUR} Sachkosten (Finanzhaushalt) zzgl. Eigenleistung des städtischen Bauhofs.
    \item \textbf{Maßnahme:} Beschaffung des Lehrtafel-Sets des Vereins \textit{Pro Igel e.V.} und Aufstellung an der Promenade.
    \item \textbf{Arbeitsauftrag an die Verwaltung:} Einplanung von Bauhof-Kapazitäten für Fundamente/Montage und Abstimmung der genauen Standorte.
    \item \textbf{Begründung:} Wertvolles und kostengünstiges Umweltbildungsangebot zum Artenschutz entlang einer stark frequentierten Flaniermeile.
\end{itemize}

\section*{4. Verlegung von „Stolpersteinen“ zur Erinnerungskultur}

\begin{itemize}
    \item \textbf{Finanzielle Auswirkungen:} \qty{2000}{\EUR} (Verwaltungshaushalt für Anschubkosten \& Recherche)
    \item \textbf{Maßnahme:} Grundsatzbeschluss zur Genehmigung der Verlegung auf städtischem Grund (Projekt Gunter Demnig); Einleitung von Recherchen zu NS-Opfern in Plattling mit Stadtarchiv und Schulen; Auflegung eines Patenschaftsmodells (\qty{120}{\EUR}--\qty{150}{\EUR} pro Stein).
    \item \textbf{Begründung:} Macht lokale Geschichte im Alltag sichtbar, stärkt das Geschichtsbewusstsein und bietet ein praxisnahes Geschichtsprojekt für Plattlinger Schulen.
\end{itemize}
