\section*{1. Anschaffung von Split-Klimageräten für städtische Gebäude}

\begin{itemize}
    \item \textbf{Finanzielle Auswirkungen:} \qty{80000}{\EUR} (Finanzhaushalt / Investitionen)
    \item \textbf{Sperrvermerk:} Qualifizierter Sperrvermerk – Die Freigabe der Haushaltsmittel erfolgt nach Vorlage und Genehmigung des Raum- und Umsetzungskonzepts im Haupt- und Finanzausschuss bzw. Bauausschuss.
    \item \textbf{Maßnahme:} Beschaffung und fachgerechte Installation von hocheffizienten Split-Klimageräten für ca. 20 besonders stark hitzebelastete Räume im städtischen Gebäudebestand.
    \item \textbf{Priorisierung der Gebäude:}
    \begin{enumerate}[label=\arabic*.]
        \item Grundschule (Schulstraße) und Mittelschule (Preysingstraße) – mit Fokus auf Dachgeschosse und nach Süden/Westen ausgerichtete Klassenzimmer.
        \item Bürgerspital (bei verbleibenden Budgetmitteln).
        \item Verwaltungsgebäude der Stadtwerke (bei weiteren Restmitteln).
    \end{enumerate}
    \item \textbf{Mögliche Förderprogramme:}
    \begin{itemize}[label=\textcolor{GruenAccent}{\small\textendash}]
        \item \textit{BAFA / KfW Bundesförderung für effiziente Gebäude (BEG EM):} Förderung von energieeffizienter Anlagentechnik bei Nutzung von Split-Geräten mit Wärmepumpenfunktion und natürlichen Kältemitteln (z.\,B. R290/Propan).
        \item \textit{Kommunalrichtlinie (NKI):} Unterstützung bei investiven Maßnahmen zum sommerlichen Wärmeschutz an Schulen.
    \end{itemize}
    \item \textbf{Arbeitsauftrag an die Verwaltung:} Ermittlung der vorrangig zu
    ertüchtigenden Unterrichtsräume in enger Abstimmung mit den jeweiligen
    Schulleitungen sowie Klärung der baulichen Voraussetzungen und
    Förderantragsfristen.
    \item \textbf{Begründung:} Extreme sommerliche Hitzeperioden beeinträchtigen
    die Konzentrations-, Lern- und Leistungsfähigkeit von Schülerinnen, Schülern
    und Lehrkräften messbar. Die Maßnahme dient dem aktiven Gesundheitsschutz
    sowie verlässlichen Arbeitsbedingungen. Durch moderne Wärmepumpentechnik
    können die Anlagen zudem in den Übergangsmonaten hocheffizient zum Heizen
    genutzt werden.
\end{itemize}

\section*{2. Mobil-flexibles Nebelsprüh-System zur Hitzeminderung}

\begin{itemize}
    \item \textbf{Finanzielle Auswirkungen:} \qty{8000}{\EUR} (Finanzhaushalt; ca. \qty{5000}{\EUR} Anschaffung, ca. \qty{3000}{\EUR} Installation, Montage und Anschlüsse)
    \item \textbf{Maßnahme:} Beschaffung eines mobil-flexiblen Nebelsprüh-Systems zur Verdunstungskühlung im öffentlichen Raum (technische Orientierung an Systemen wie \textit{Water Oasis} in Fürth).
    \item \textbf{Standortkonzept:} 
    \begin{itemize}[label=\textcolor{GruenAccent}{\small\textendash}]
        \item \textit{Hauptstandort:} Spielplatz Preysingstraße / Sterngasse (dicht bebauter Bereich).
        \item \textit{Mobile Nutzung:} temporäre Umsetzung zu innerstädtischen Veranstaltungen (z.\,B. Volksfestplatz, Nibelungenfest)
    \end{itemize}
    \item \textbf{Mögliche Förderprogramme:}
    \begin{itemize}[label=\textcolor{GruenAccent}{\small\textendash}]
        \item \textit{KommKlimaFöR 2026 (Freistaat Bayern):} Explizite Förderung von investiven Klimaanpassungsmaßnahmen wie „klimafreundlichen Wasseranlagen zur Kühlung der Umgebung“ auf öffentlichen Plätzen und Spielplätzen. Reguläre Förderquote bis zu \SI{50}{\percent} (bei Bedarf bis zu \SI{90}{\percent}).
    \end{itemize}
    \item \textbf{Arbeitsauftrag an die Verwaltung:} Technische Prüfung der örtlichen Wasseranschluss am Spielplatz sowie Einreichung des Förderantrags über die Fördermanagementplattform Bayern (FMP) vor Auftragsvergabe.
    \item \textbf{Begründung \& Schutzwirkung:} Feine Hochdruck-Verdunstungskühlung senkt die gefühlte Umgebungstemperatur an Hitzetagen spürbar ab, ohne Nässe zu erzeugen, und benötigt dabei nur minimale Wassermengen. Die Anlage schützt im Bereich des Spielplatzes täglich zahlreiche Kinder und Eltern vor akuter Hitzebelastung.
\end{itemize}

\section*{3. Errichtung eines Igel-Lehrpfades an der Mühlbach-Promenade}

\begin{itemize}
    \item \textbf{Finanzielle Auswirkungen:} \qty{1000}{\EUR} Sachkosten (Finanzhaushalt) für Beschilderung und Infotafeln zuzüglich der bauseitigen Eigenleistung durch den städtischen Bauhof.
    \item \textbf{Maßnahme:} Beschaffung des praxiserprobten, kindgerechten Lehrtafel-Sets des Vereins \textit{Pro Igel e.V.} und dauerhafte Aufstellung entlang der Mühlbach-Promenade.
    \item \textbf{Mögliche Förderprogramme:}
    \begin{itemize}[label=\textcolor{GruenAccent}{\small\textendash}]
        \item \textit{Bayerischer Naturschutzfonds / Umweltstiftungen:} Förderung lokaler Projekte zur Natur- und Artenschutzbildung im Siedlungsraum.
        \item \textit{ILE-Regionalbudget / Kleinprojekte:} Bis zu \SI{80}{\percent} Fördersatz für gemeinwohlorientierte Kleinmaßnahmen im ländlichen Raum.
    \end{itemize}
    \item \textbf{Arbeitsauftrag an die Verwaltung:} Einplanung der benötigten Bauhof-Kapazitäten für Fundamente, Pfosten und Montagearbeiten sowie Festlegung der genauen Tafelstandorte entlang des Spazierwegs.
    \item \textbf{Begründung:} Der Igel steht als Leitart für einen artenreichen Garten- und Siedlungsraum, gerät jedoch zunehmend unter Druck. Ein Igel-Lehrpfad schafft mit minimalem finanziellem Aufwand ein dauerhaftes, interaktives Umweltbildungsangebot für Spaziergänger, Familien, Kindergärten und Schulklassen entlang einer der meistfrequentierten Naherholungsachsen Plattlings.
\end{itemize}

\section*{4. Verlegung von „Stolpersteinen“ zur Erinnerungskultur}

\begin{itemize}
    \item \textbf{Finanzielle Auswirkungen:} \qty{2000}{\EUR} (Verwaltungshaushalt als Anschubfinanzierung für wissenschaftliche/lokalhistorische Recherche und organisatorische Vorbereitung).
    \item \textbf{Maßnahme:} 
    \begin{enumerate}[label=\arabic*.]
        \item Grundsatzbeschluss des Stadtrats zur Genehmigung der Verlegung von Messing-Gedenksteinen auf städtischem Grund (im Rahmen des europaweiten Projekts von Gunter Demnig).
        \item Einleitung und Begleitung von Recherchen zu NS-Opfern mit Bezug zu Plattling in Kooperation mit dem Stadtarchiv, Historiker*innen und den weiterführenden Schulen.
        \item Einrichtung eines transparenten Patenschaftsmodells zur Finanzierung der eigentlichen Steine (\qty{120}{\EUR} bis \qty{150}{\EUR} je Stein).
    \end{enumerate}
    \item \textbf{Refinanzierung:} Die Anschaffungs- und Verlegekosten der einzelnen Steine werden vollumfänglich über private Patenschaften von Bürgerinnen und Bürgern, Vereinen, Parteien, Firmen oder Schulen gegen Spendenbescheinigung getragen.
    \item \textbf{Begründung:} Stolpersteine verankern das Gedenken an Opfer von Verfolgung und Gewaltherrschaft mitten im alltäglichen Lebensraum. Sie machen Regionalgeschichte greifbar, wirken geschichtsrevisionistischen Tendenzen entgegen und bieten Jugendlichen eine wertvolle Plattform für praxisnahe historische Projektarbeit vor Ort.
\end{itemize}
